\section{Literature}
%----------------------------------------------------------------------
\begin{frame}{Why fiscal non-tradables?}
  \begin{block}{Core question \paper}
    Government spending falls disproportionately on \emph{non-tradable services}.
    Empirically, a spending shock still raises private consumption, raises goods
    \emph{and} services output, \emph{lowers} the relative price of goods, and
    \emph{worsens} net exports.
  \end{block}
  \begin{itemize}
    \item A two-sector open RANK with the same $G$ bias does \emph{not} deliver
          those co-movements---especially when $G$ is purely non-tradable.
    \item The paper asks whether \textbf{household heterogeneity} plus
          \textbf{trade openness} can generate the consumption-led spillover
          from fiscal non-tradables into the goods sector.
    \item That is a statement about \textbf{sectoral fiscal multipliers} in an
          open economy, not about a one-sector closed-economy multiplier.
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Open-economy HANK, fiscal--monetary, T vs.\ NT}
  \begin{itemize}
    \item \textbf{Fiscal HANK.}
          Incomplete markets and high iMPCs reverse RANK crowding-out of $C$
          (Kaplan, Moll and Violante 2018; Auclert, Rognlie and Straub on the
          intertemporal Keynesian cross and iMPCs). Sticky information
          (Auclert, Rognlie and Straub 2020) turns jumps into humps---used
          here with $\vartheta=0.935$.
    \item \textbf{Open-economy HANK.}
          Heterogeneous agents change exchange-rate and trade transmission
          (Auclert, Rognlie, Souchier and Straub; de Ferra, Mitman and Romei).
          Fiscal leakage through imports is no longer a representative-agent
          Euler residual.
    \item \textbf{Sectoral T / NT.}
          Distribution costs (Corsetti and Dedola 2005) and imperfect labour
          mobility (Horvath) break wage pass-through from NT into tradable
          prices. Composition of $G$ then matters for relative prices.
    \item \textbf{This talk.}
          Paper first (S-BVAR facts, two-sector open HANK, claims), then an
          honest status report on the open-source Figure~3 prototype---not
          the authors' files.
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{RANK vs.\ HANK fiscal transmission}
  \begin{itemize}
    \item \textbf{RANK.}
          Ricardian households. A financed rise in $G$ raises the present
          value of taxes and typically \emph{crowds out} private consumption.
          In an open RANK, expenditure switching and import leakage can move
          NX, but the consumption-led spillover to tradables is missing when
          $G$ itself has little tradable content.
    \item \textbf{HANK.}
          Uninsurable income risk and a borrowing constraint $a\ge 0$ produce
          a mass of high-iMPC households. Labour income from NT production
          raises $C$ on impact; CES demand then spills into goods.
    \item \textbf{Openness is not a sideshow.}
          Import intensity and limited labour reallocation dampen wage
          pass-through to tradable prices. Relative goods prices can
          \emph{fall} even as $Y_T$ rises; NX can deteriorate while $Y_T$
          still co-moves positively (expenditure switching).
    \item \textbf{Paper's identification of the mechanism.}
          Nested RANK vs.\ HANK, and a $\phi_{GT}=0$ (pure NT) counterfactual:
          HANK keeps $Y_T>0$; RANK does not.
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Four stylised facts (S-BVAR)}
  Empirical block (WP \S2): Blanchard--Perotti identification, U.S.\
  1954Q1--2019Q4, nine variables, four lags, NIW prior.
  Shock: orthogonal \textbf{1\% of output} increase in $G$.
  \begin{enumerate}
    \item Private consumption \textbf{rises} (crowding-in, not RANK crowding-out).
    \item Output rises in \textbf{goods and services}.
    \item The relative price of goods \textbf{falls}.
    \item Net exports \textbf{deteriorate}.
  \end{enumerate}
  \vspace{0.4em}
  {\footnotesize
  The theoretical $G$ path is the empirical IRF (``does not follow a hump-shaped
  path'', \S4.3). This repo uses a 20-quarter path digitised from Figure~3's
  $G$ panel \textbf{[PROVISIONAL]}, not the authors' series.}
\end{frame}

\section{The model}
%----------------------------------------------------------------------
\begin{frame}{Two-sector open HANK at a glance}
  \begin{itemize}
    \item Domestic production: \textbf{tradables} ($T$, goods) and
          \textbf{non-tradables} ($NT$, services). Linear technology
          $Y_S=N_S$; flexible goods prices, sticky wages only (baseline).
    \item Incomplete-markets households; hours set by sectoral unions
          ($n_{S,t}(i)=N_{S,t}$). Nested RANK counterpart: same aggregates,
          Euler instead of the HA Jacobian.
    \item Fiscal $G$ is CES-split with $\phi_{GT}=0.25$ (OECD COFOG tradable
          content); HA-LB / RA-LB set $\phi_{GT}=0$.
    \item Small open economy: CES home/foreign, Corsetti--Dedola distribution,
          UIP with an NFA wedge, tax rule on debt, Taylor rule on CPI.
    \item Solution: perfect-foresight MIT shocks, sequence-space Jacobians,
          linearised (Auclert, Bard\'oczy, Rognlie and Straub 2021).
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Households (WP \S3.1)}
  \begin{itemize}
    \item Incomplete markets, $a\ge 0$, idiosyncratic $e_t$ AR(1)
          ($\rho_e=0.968$). Post-tax labour income
          \[
            z_t(i)=(w_T N_T+w_{NT}N_{NT}-T_t)\,e_t(i).
          \]
    \item Utility (4) with $\sigma=\varphi=1$. Households choose $c,a$ only;
          hours are union-set.
    \item CES demands: $T$ vs.\ $NT$ ($\lambda_t=2.15$, $\phi_t=0.623$);
          home vs.\ foreign with trade elasticity $\lambda_d=1$
          (Cobb--Douglas; 12\% import/GDP via $\phi_d=0.57$).
    \item Sticky expectations, share $\vartheta=0.935$ that does not update
          (paper eq.\ 13 / Auclert--Rognlie--Straub mapping).
    \item Nested RANK: same budget, representative Euler, same $r$
          ($\beta_{RA}=0.995$ vs.\ $\beta_{HA}=0.986$).
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Production, labour, distribution}
  \begin{itemize}
    \item Flexible prices: $P_Z=W_S$ (eq.\ 14). Sticky-price Appendix~C is
          not in the baseline DAG.
    \item Imperfect labour mobility: Horvath aggregator (9), $\lambda_l=1$.
          Sectoral wage Phillips curves (12), slope $\kappa_w=0.0044$
          (Lind\'e et al.\ 2016), $\theta_w=3$ (50\% markup).
    \item Distribution (Corsetti--Dedola, eqs.\ 28--31): local NT services
          are bundled into retail prices of domestic and imported tradables
          ($\mu_d=0.266$, $\mu_f=0.754$). That is the sectoral \emph{cost}
          channel from NT wages into $T$ prices.
    \item Output identities (34)--(35): $Y_T$ is domestic tradables plus
          exports; $Y_{NT}$ is services plus distribution margins.
  \end{itemize}
  \vfill
  {\scriptsize Seminar reading: limited labour reallocation plus distribution
  costs let NT demand raise $Y_{NT}$ without immediately bidding up goods
  prices one-for-one---the relative $T/NT$ price can fall.}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Fiscal, monetary, open economy}
  \begin{align}
    T_t &= T_{ss}+\kappa_b(B_{t-1}-B_{ss})
      \tag{tax, 21} \\[0.2em]
    i_t &= \zeta\, i_{t-1}+(1-\zeta)(i^\ast+\kappa_\pi\pi_t)
      \tag{Taylor, 22}
  \end{align}
  \begin{itemize}
    \item $G$ exogenous; CES split (16)--(19) with $\phi_{GT}=0.25$,
          $\phi^G_d=1$. Budget (20). $\kappa_b=0.027$, $\zeta=0.9$,
          $\kappa_\pi=1.5$.
    \item Mutual fund / UIP (23)--(26). Paper:
          $\Gamma_t=\exp(-10^{-9}\,\mathrm{nfa}_t)$.
    \item Exports $X_t=C^\ast_{F,t}/(1+\mu_f)$; imports through the
          distribution wedge. Goods/asset clearing (36)--(37).
    \item SSJ DAG: unknowns $(Y_T,Y_{NT},w_T,w_{NT},\pi,Q,i,B,\mathrm{nfa})$;
          $\mathtt{goods\_NT}$ dropped by Walras (household + government +
          NFA laws of motion).
  \end{itemize}
\end{frame}

\section{Contributions}
%----------------------------------------------------------------------
\begin{frame}{What the NBB WP adds}
  Relative to one-sector HANK and to open-economy RANK fiscal:
  \begin{itemize}
    \item An \textbf{empirical} four-fact package: crowding-in of $C$,
          positive goods \emph{and} services output, a decline in the
          relative goods price, and a deterioration of NX after a $G$ shock
          that is concentrated in NT services.
    \item A \textbf{two-sector open HANK} in which uninsurable income risk
          and precautionary saving produce a persistent income-driven
          expansion in $C$; import intensity and limited labour reallocation
          dampen wage pass-through to goods prices.
    \item \textbf{Expenditure switching} then yields $Y_T>0$ even as NX
          worsens---and even when $\phi_{GT}=0$. Nested RANK fails that
          test.
    \item Quantitative message: \textbf{household heterogeneity and trade
          openness jointly} shape sectoral fiscal multipliers. Composition
          of $G$ is not a sideshow.
  \end{itemize}
\end{frame}

\begin{frame}{Mechanism, and what is \emph{not} claimed}
  \begin{itemize}
    \item \textbf{HANK mechanism (paper).}
          NT-biased $G$ $\to$ labour income $\to$ high iMPC consumption
          $\to$ CES demand for goods $\to$ $Y_T>0$ even if $\phi_{GT}=0$.
          Distribution + imperfect mobility: $P_T/P_{NT}$ can fall.
          Openness: NX deteriorates while expenditure switching supports
          $Y_T$.
    \item \textbf{RANK contrast.}
          $C$ falls. With $\phi_{GT}=0.25$, $Y_T$ can still rise from the
          tradable content of $G$; with $\phi_{GT}=0$, that co-movement
          disappears.
    \item \textbf{Known shape gaps vs.\ the published figure}
          (NOTES.md; this code):
          HA $C$ is a bit too front-loaded (sticky-info hump milder);
          HA $Y_T$ does not reproduce the empirical hump;
          HA NX stays negative too long (the authors note a medium-run NX
          undershoot);
          RANK NX is a small surplus rather than a small deficit.
  \end{itemize}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{Why this matters for recent fiscal / open-economy debates}
  \begin{itemize}
    \item \textbf{Composition of $G$.}
          Much public spending is services (health, education,
          administration). Multipliers quoted on aggregate $G$ hide a
          T / NT split; the paper's $\phi_{GT}=0.25$ is the empirically
          relevant bias, not a knife-edge.
    \item \textbf{Crowding-in vs.\ crowding-out.}
          Post-COVID fiscal packages revived the RANK prediction that $C$
          should fall. The S-BVAR sample here ends in 2019Q4; the
          \emph{mechanism}---high iMPCs plus NT-biased $G$---is the relevant
          object for those debates, not a COVID-period estimate.
    \item \textbf{Openness and leakage.}
          Import intensity need not kill the goods-sector multiplier:
          expenditure switching can raise $Y_T$ even as NX worsens.
          Closed-economy HANK misses that joint restriction.
    \item \textbf{Relative prices.}
          A fiscal expansion that is NT-heavy can \emph{lower} goods prices
          relative to services---a sectoral relative-price fact, not a
          one-sector inflation multiplier.
  \end{itemize}
\end{frame}

\begin{frame}{Calibration knobs and caveats}
  \begin{itemize}
    \item \textbf{$\sigma_e$.}
          Table~2 lists $0.58$. Used as the SSJ Rouwenhorst unconditional SD
          of $\log e$, that misses Table~1 ($iMPC_1\approx 0.24$, HtM $\approx 20\%$).
          Default here: \texttt{-{-}sigma-e 0.35}, which hits Table~1
          $iMPC_1\approx 0.51$ and HtM $\approx 49.7\%$ given $\beta_{HA}$
          and $r=1/\beta_{RA}-1$ \textbf{[PROVISIONAL]}.
    \item \textbf{UIP wedge.}
          Paper $\psi_{\mathrm{nfa}}=10^{-9}$; CLI default
          \texttt{-{-}psi-nfa 1e-3} for a stable linear solve
          \textbf{[PROVISIONAL]}.
    \item \textbf{S-BVAR overlay} in \texttt{empirical\_sbvar.py}: hand-read
          from Figure~3, not the authors' series \textbf{[PROVISIONAL]}.
    \item \textbf{Other closures} (NOTES.md): $G/Y\approx 0.214$ implied by
          12\% import/GDP; $\phi_l$ set to sectoral hours shares so both
          wage PCs hold at $w=1$; distribution margin on $G_D$ for labour
          income $=$ GDP; SOE export demand with $C^\ast$ fixed; SSJ
          ex-post $r=$ lagged ex-ante.
  \end{itemize}
\end{frame}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item NT-biased government spending can still raise $C$, raise $Y_T$
          \emph{and} $Y_{NT}$, lower $P_T/P_{NT}$, and worsen NX---if
          households are incomplete-markets and the economy is open.
    \item Nested RANK fails that package, especially when $\phi_{GT}=0$.
          Heterogeneity, not the tradable content of $G$, is the spillover.
    \item This repo is a transparent Figure~3 starter: Python + SSJ DAG,
          qualitative HANK vs.\ RANK signs, documented provisional
          $\sigma_e$ / UIP / S-BVAR closures---not a numerical replica of
          the authors' IRFs.
    \item The natural extension is a labelled multi-sector open DAG, not a
          one-file N-sector solver.
  \end{itemize}
